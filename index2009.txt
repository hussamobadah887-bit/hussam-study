function listenChat(){
 if(chatUnsub)chatUnsub();
 if(!currentUser)return;
 try{
  const q=query(collection(db,"chat"),orderBy("createdAt","asc"));
  chatUnsub=onSnapshot(q,snap=>{
   const box=$("messages");if(!box)return;
   if(snap.empty){box.innerHTML='<div class="empty">لا توجد رسائل بعد.</div>';return}
   box.innerHTML="";
   snap.forEach(d=>{
    const m=d.data(), mine=m.uid===currentUser.uid;
    const row=document.createElement("div");
    row.className="msg"+(mine?" me":"");
    row.innerHTML=`<div class="msg-head">
      <div class="name">${esc(m.name||"طالب")}</div>
      ${mine?`<button class="chat-delete" onclick="deleteChatMessage('${d.id}')">🗑️ حذف</button>`:""}
    </div><div>${esc(m.text||"")}</div>`;
    box.appendChild(row);
   });box.scrollTop=box.scrollHeight;
  },e=>{const s=$("chatStatus");if(s)s.textContent="تعذر تحميل الشات: "+(e.message||"")});
 }catch(e){}
}
window.sendChat=async()=>{
 const input=$("chatInput"),text=input?.value.trim(),status=$("chatStatus");
 if(!text)return;
 if(!currentUser){if(status)status.textContent="سجّل الدخول أولاً.";return}
 try{
  const p=getProfile();
  await addDoc(collection(db,"chat"),{
    uid:currentUser.uid,
    name:p.name||currentUser.displayName||"طالب",
    photo:p.photo||currentUser.photoURL||"",
    text,
    createdAt:serverTimestamp()
  });
  input.value="";input.focus();if(status)status.textContent="تم الإرسال ✓";
 }catch(e){if(status)status.textContent="تعذر الإرسال. تأكد من صلاحيات Firestore لمجموعة chat."}
};
window.deleteChatMessage=async(id)=>{
 if(!currentUser)return;
 try{
  // Dynamic import keeps the initial Firebase import small.
  const mod=await import("https://www.gstatic.com/firebasejs/12.19.0/firebase-firestore.js");
  const ref=mod.doc(db,"chat",id);
  const snap=await mod.getDoc(ref);
  if(!snap.exists())return;
  if(snap.data().uid!==currentUser.uid){
    const s=$("chatStatus");if(s)s.textContent="لا يمكنك حذف رسالة شخص آخر.";return;
  }
  await mod.deleteDoc(ref);
  const s=$("chatStatus");if(s)s.textContent="تم حذف رسالتك ✓";
 }catch(e){
  const s=$("chatStatus");if(s)s.textContent="تعذر حذف الرسالة. تأكد من صلاحيات Firestore.";
 }
};

function aiReply(text){
 const t=text.toLowerCase();
 const all=Object.entries(data).flatMap(([k,a])=>a.map(x=>({...x,key:k})));
 if(t.includes("تقدم")||t.includes("نسبة")){
  const lines=Object.entries(subjects).map(([k,n])=>{
   const a=data[k],d=a.filter(x=>x.done).length,p=a.length?Math.round(d/a.length*100):0;
   return `${n}: ${p}%`;
  });
  return "تقدمك حسب كل مادة:\n"+lines.join("\n");
 }
 if(t.includes("إنجاز")||t.includes("انجاز")){
  const open=all.filter(x=>!x.done).slice(0,5);
  return open.length?"الإنجازات غير المكتملة عندك:\n"+open.map(x=>`• ${subjects[x.key]}: ${x.text}`).join("\n"):"لا يوجد لديك إنجازات غير مكتملة.";
 }
 if(t.includes("خطة")||t.includes("اليوم")){
  const open=all.filter(x=>!x.done).slice(0,3);
  return open.length?"ابدأ اليوم بهذه الإنجازات:\n"+open.map(x=>`1. ${x.text}`).join("\n"):"أضف إنجازات أولاً حتى أساعدك بخطة.";
 }
 return "وصلتني رسالتك: «"+text+"»\n\nأقدر أساعدك في متابعة إنجازات المواد وحساب تقدمك وتنظيم دراستك.";
}
